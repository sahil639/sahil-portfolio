An inline image floats to the side so text can wrap around it on desktop. On phones it goes full width. Keep writing here and the paragraph flows next to the picture, which is nice for small details, screenshots of a single button, or a portrait photo next to a quote.

export function spring(from, to, t) {
  const k = 170, d = 26; // highlighted
  return to + (from - to) * Math.exp(-d * t) * Math.cos(k * t); // highlighted
}
